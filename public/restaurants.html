async function loadShops() { try { // 1. Get the category from the URL (e.g. /restaurants.html?category=Pharmacy) const params = new URLSearchParams(window.location.search); const category = params.get('category'); let url = '/api/catalog/shops'; if (category) { url += `?category=${encodeURIComponent(category)}`; } const response = await fetch(url); const data = await response.json(); // Ensure we handle both array responses and object responses shops = Array.isArray(data) ? data : (data.shops || []); drawShops(); } catch (e) { JD.fail(e, $('shops')); } }
